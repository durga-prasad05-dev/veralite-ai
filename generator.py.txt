from openai import OpenAI
from prompts import build_prompt
from validator import validate_message

client = OpenAI(api_key="YOUR_API_KEY")


SYSTEM_PROMPT = """
You are Vera, India’s smartest merchant engagement AI.

Generate highly natural WhatsApp-style promotional messages.

Rules:
- Sound human
- Avoid robotic language
- Keep messages short
- Mention real offers
- Create urgency naturally
- Use conversational Indian style
"""


def generate_message(data):
    prompt = build_prompt(data)

    response = client.chat.completions.create(
        model="gpt-4.1-mini",
        messages=[
            {
                "role": "system",
                "content": SYSTEM_PROMPT
            },
            {
                "role": "user",
                "content": prompt
            }
        ],
        temperature=0.9
    )

    message = response.choices[0].message.content.strip()

    score = validate_message(message)

    return {
        "message": message,
        "score": score
    }

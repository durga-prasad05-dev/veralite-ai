import json
from generator import generate_message


def main():
    print("\n=== VeraLite AI Merchant Message Engine ===\n")

    try:
        with open("sample_data.json", "r", encoding="utf-8") as file:
            data = json.load(file)

        result = generate_message(data)

        print("Generated Message:\n")
        print(result["message"])

        print("\nQuality Score:")
        print(result["score"])

    except FileNotFoundError:
        print("Error: sample_data.json file not found.")

    except json.JSONDecodeError:
        print("Error: Invalid JSON format in sample_data.json")

    except Exception as e:
        print("Unexpected Error:", e)


if __name__ == "__main__":
    main()
